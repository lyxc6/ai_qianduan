            { name: "练体期", exp: 0, nextExp: 100, maxMp: 100, maxSlots: 1 },

        function calcCorrectStage(expValue) {
            let correctStage = STAGE_MAP[0];
            for (let i = 0; i < STAGE_MAP.length; i++) {
                if (expValue >= STAGE_MAP[i].exp) {
                    correctStage = STAGE_MAP[i];
                } else {
                    break;
                }
            }
            return correctStage;
        }

        function getSafeMessageId() {
            if (typeof window.parent.Mvu === 'undefined') return null;
            let msgId = null;
            if (window.parent.getCurrentMessageId) {
                msgId = window.parent.getCurrentMessageId();
            }
            if (msgId === null || msgId === undefined) {
                 if (window.parent.SillyTavern && window.parent.SillyTavern.getContext) {
                     const ctx = window.parent.SillyTavern.getContext();
                     if (ctx && ctx.chat && ctx.chat.length > 0) {
                         msgId = ctx.chat.length - 1;
                     }
                 }
            }
            return msgId;
        }

        async function syncCoreStats() {
            if(isSyncingStats) return;
            const p = window.parent;
            if(!p || !p.Mvu) return;

            const msgId = getSafeMessageId();
            if(msgId === null || msgId === undefined) return;

            const rawVars = p.Mvu.getMvuData({ type: 'message', message_id: msgId });
            if(!rawVars || !rawVars.stat_data || !rawVars.stat_data["世界与本源"]) return;

            const core = rawVars.stat_data["世界与本源"];
            const currentExp = Number(core["灵力值"] || 0);
            const currentStageName = core["境界"] || "练体期";
            const correctStageInfo = calcCorrectStage(currentExp);

            const bodies = rawVars.stat_data["夺舍女躯"] || {};
            // 这里去掉了原本强制本体+1的逻辑
            // 真实占用名额完全等于存活的女躯数量
            const actualSlots = Object.keys(bodies).length;

            const currentSlots = Number(core["已用夺舍名额"] || 0);
            let needSave = false;

            if(correctStageInfo.name !== currentStageName) {
                core["境界"] = correctStageInfo.name;
                needSave = true;
                toastr.success(`阳气满溢，灵台升华！神魂已自动突破至【${correctStageInfo.name}】！`);
            }

            if(actualSlots !== currentSlots) {
                core["已用夺舍名额"] = actualSlots;
                needSave = true;
            }

            if(needSave) {
                isSyncingStats = true;
                try {
                    await p.Mvu.replaceMvuData(rawVars, { type: 'message', message_id: msgId });
                    updateUI();
                } catch(e) {
                    console.error("核心数值自动同步失败", e);
                } finally {
                    isSyncingStats = false;
                }
            }
        }

        async function init() {
            await waitGlobalInitialized('Mvu');
            updateUI();
            syncCoreStats();

            eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, () => {
                updateUI();
                syncCoreStats();
            });

            $('.tab-btn').on('click', function() {
                const idx = $(this).data('tab');
                $('.tab-btn').removeClass('active');
                $(this).addClass('active');
                $('.tab-content').removeClass('active');
                $(`#tab${idx}`).addClass('active');
            });

            $('#sel-body').on('change', renderBodyDetail);
            $('#sel-girl').on('change', renderBoyDetail);

            $('.tag-btn').on('click', function() {
                const actionStr = $(this).data('action');
                let execName = $('#sel_executor').val() || '{{user}}';
                let finalText = execName + actionStr;
                try {
                    const p = window.parent;
                    if (p && p.$) {
                        const $textarea = p.$('#send_textarea');
                        if ($textarea.length) {
                            let currentVal = $textarea.val();
                            let newVal = currentVal.length > 0 ? currentVal + '\\n' + finalText : finalText;
                            $textarea.val(newVal).trigger('input');
                        }
                    }
                } catch (e) { console.error("输入交互失败:", e); }
            });

            // 暴力采补模式切换逻辑
            $(document).on('click', '#btn-toggle-brutal', async function() {
                const btn = $(this);
                if (btn.prop('disabled')) return;
                btn.text('切换中...').prop('disabled', true);

                try {
                    const msgId = getSafeMessageId();
                    if(msgId === null || msgId === undefined) throw new Error("无法获取当前楼层ID");
                    const rawVars = Mvu.getMvuData({ type: 'message', message_id: msgId });
                    if(!rawVars || !rawVars.stat_data) throw new Error("无法获取变量数据");

                    let currentStatus = _.get(rawVars.stat_data, '世界与本源.暴力采补模式', false);
                    let newStatus = !currentStatus;
                    _.set(rawVars.stat_data, '世界与本源.暴力采补模式', newStatus);

                    await Mvu.replaceMvuData(rawVars, { type: 'message', message_id: msgId });

                    updateUI();

                    if (newStatus) {
                        toastr.success(`【暴力采补】已开启！男鼎将被疯狂榨干至死，灵力翻倍！`);
                    } else {
                        toastr.info(`已切回【常规采补】，细水长流不伤本源。`);
                    }
                } catch (e) {
                    console.error("暴力模式切换失败:", e);
                    toastr.error("模式切换失败，请检查运行环境");
                } finally {
                    btn.text('切换手段').prop('disabled', false);
                }
            });

            $(document).on('click', '.toggle-auto-btn', async function() {
                const btn = $(this);

                const bodyName = btn.data('name');
                btn.text('切换中...').prop('disabled', true);

                try {
                    const msgId = getSafeMessageId();
                    if(msgId === null || msgId === undefined) throw new Error("无法获取当前楼层ID");

                        toastr.warning("当前女躯暂无详细数据");

                        return;

                    let newStatus = !currentStatus;
                    _.set(rawVars.stat_data, `夺舍女躯.${bodyName}.是否自主运行`, newStatus);

                    await Mvu.replaceMvuData(rawVars, { type: 'message', message_id: msgId });
                    updateUI();

                    if (newStatus) {
                        toastr.success(`已开启【${bodyName}】的自主挂机！她将按原主人设自主行动。`);

            $(document).on('click', '.thought-btn', async function() {
                const btn = $(this);
                const boyName = btn.data('name');
                btn.text('施法中...').prop('disabled', true);

                try {
                    const msgId = getSafeMessageId();
                    if(msgId === null || msgId === undefined) throw new Error("无法获取当前楼层ID");
                    const rawVars = Mvu.getMvuData({ type: 'message', message_id: msgId });

                    await Mvu.replaceMvuData(rawVars, { type: 'message', message_id: msgId });

                    unlockedThoughts.add(boyName);

                    toastr.success(`成功窥探男鼎【${boyName}】的心声`);
                } catch(e) {
                    console.error("窥心术施展失败:", e);

                    btn.text('消耗 10 法力窥探此男心声').prop('disabled', false);
                }
            });

            $(document).on('click', '.btn-del-boy', async function() {
                const btn = $(this);
                const boyName = btn.data('name');

                if(!confirm(`确定要将男鼎【${boyName}】从记录中彻底抹除吗？（删除后不可恢复）`)) {
                    return;
                }

                    const msgId = getSafeMessageId();
                    if(msgId === null || msgId === undefined) throw new Error("无法获取当前楼层ID");
                    const rawVars = Mvu.getMvuData({ type: 'message', message_id: msgId });

                    if(rawVars && rawVars.stat_data && rawVars.stat_data["男性鼎炉录"] && rawVars.stat_data["男性鼎炉录"][boyName]) {

                        updateUI();

                        toastr.warning("未找到该男鼎的数据");

                    }
                } catch(e) {
                    console.error("删除失败:", e);
                    toastr.error("删除失败");
                    btn.text('☠ 彻底抹除此鼎炉').prop('disabled', false);
