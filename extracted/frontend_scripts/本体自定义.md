        function getSafeMessageId() {
            if (typeof window.parent.Mvu === 'undefined') return null;
            let msgId = null;
            if (window.parent.getCurrentMessageId) msgId = window.parent.getCurrentMessageId();
            if (msgId === null || msgId === undefined) {
                 if (window.parent.SillyTavern && window.parent.SillyTavern.getContext) {
                     const ctx = window.parent.SillyTavern.getContext();
                     if (ctx && ctx.chat && ctx.chat.length > 0) msgId = ctx.chat.length - 1;
                 }
            }
            return msgId;
        }

        async function lockWorld() {
            var selVal = document.getElementById('sel_world_type').value;
            var finalWorldName = selVal;
            var finalWorldIntro = document.getElementById('ipt_world_intro').value.trim();

            if (selVal === '自定义世界观') {
                var customVal = document.getElementById('ipt_custom_world').value.trim();
                if (!customVal) { alert("哥哥，请输入自定义世界观的名称哦！"); return; }
                if (!finalWorldIntro) { alert("哥哥，自定义世界观需要填写简单的世界简介哦！"); return; }
                finalWorldName = customVal;
                currentActiveWorld = "现代都市";
            } else {
                currentActiveWorld = selVal;
            }

            try {
                const msgId = getSafeMessageId();
                if (msgId !== null && typeof msgId !== 'undefined') {
                    const rawVars = window.parent.Mvu.getMvuData({ type: 'message', message_id: msgId });
                    if (!rawVars.stat_data) rawVars.stat_data = {};
                    if (!rawVars.stat_data["世界与本源"]) rawVars.stat_data["世界与本源"] = {};

                    rawVars.stat_data["世界与本源"]["当前世界观"] = finalWorldName;
                    rawVars.stat_data["世界与本源"]["世界简介"] = finalWorldIntro;
                    await window.parent.Mvu.replaceMvuData(rawVars, { type: 'message', message_id: msgId });
                }
            } catch(e) { console.error("MVU写入失败：", e); }

            document.getElementById('sel_world_type').disabled = true;
            document.getElementById('ipt_custom_world').disabled = true;
            document.getElementById('ipt_world_intro').readOnly = true;
            var btn = document.getElementById('btn_lock_world');
            btn.innerText = "世界已锚定";
            btn.disabled = true;

            fillDynamicOptions();
            document.getElementById('main_creation_area').style.display = 'block';
        }
